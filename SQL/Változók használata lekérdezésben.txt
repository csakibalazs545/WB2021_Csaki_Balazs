Változók használata lekérdezésben
A következő SELECTutasítás  @model_yeara záradékban található változót használja WHEREegy adott modellév termékeinek megkereséséhez:

SELECT
    product_name,
    model_year,
    list_price 
FROM 
    production.products
WHERE 
    model_year = @model_year
ORDER BY
    product_name;
